public class Ticket
{

    /// 0 - regular
    /// 1 - bronze
    /// 2 - silver
    /// 3 - gold
    /// </summary>
    public int type;

    /// <summary>
    /// 0-marked
    /// 1-non marked regular
    /// 2-collectable
    /// 3-spread
    /// 4-instant bingo
    /// </summary>
    public List<int> cells = new List<int>();

    /// <summary>
    /// List(25)
    /// </summary>
    public List<int> cellNumbers = new List<int>();
    
    /// <summary>
    /// Mult * averageBet
    /// </summary>
    public long prize;

    public int collectedItems = 0;

    public List<int> specialCombination = new List<int>();
}
